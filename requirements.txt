# =====================================================================
# voxceleb_trainer / SL_ColvaiAI runtime dependencies
# Pinning policy: major-version-bounded for the PyTorch ecosystem and
# scientific stack so unattended `pip install -r requirements.txt`
# cannot pick up a breaking major release. See:
#   docs/bugfixes/BUGFIX-011-requirements-pins.md
# =====================================================================

# PyTorch ecosystem ---------------------------------------------------
# Floor: 2.1 (matches the team's actual training env — PyTorch 2.9.0
# per research_logs/2025-10-20.md — and captures every modern API the
# code uses: register_buffer(..., persistent=False) [>=1.6],
# zero_grad(set_to_none=True) [>=1.7], torch.inference_mode() [>=1.9],
# GradScaler(enabled=...) [>=1.6]). Floor also avoids the NumPy-2
# incompatibility window in torch <2.3.
# Ceiling: <3 to prevent silent upgrade to a hypothetical breaking 3.0.
# torchaudio must match torch's major version per the PyTorch
# release-pairing matrix.
torch>=2.1,<3
torchaudio>=2.1,<3

# Audio / signal processing -------------------------------------------
# asteroid_filterbanks is pinned exactly: RawNet3.py constructs
# ParamSincFB with a 0.4.0-specific keyword signature.
asteroid_filterbanks==0.4.0

# scipy.signal.resample_poly / fftconvolve / convolve and
# scipy.io.wavfile are the consumed APIs; all stable since 1.7.
scipy>=1.7,<2

# soundfile.read(..., dtype='float32', always_2d=...) — stable API.
soundfile>=0.10,<1

# Numerical / ML utilities --------------------------------------------
# NumPy 2.x is supported after the 2025-10-20 compatibility pass
# (numpy.float -> float migration documented in research_logs/2025-10-20.md).
numpy>=1.20,<3

# Only sklearn.metrics.roc_curve is used; floor is conservative.
scikit-learn>=1.0,<2

# Used by dataprep.py progress bars and similar.
tqdm>=4.0,<5

# Config parsing.
pyyaml>=5.0,<7

# Optional (training-time observability) ------------------------------
# Imported via try/except in trainSpeakerNet*.py; the trainer falls
# back gracefully if absent. Uncomment to enable TensorBoard scalar
# logging and ROC-curve plots in the trainer's best-model branch.
# tensorboard>=2.0,<3
# matplotlib>=3.0,<4

# Optional (SSL front-end via FEATURE-001) ----------------------------

# FEATURE-001 (model: MLPMixerSpeaker_RawWaveform) does not need this.
# 4.30 is the floor that ships a stable AutoModel API for the tested
